📖 API 文档

全局对象

game - 游戏主对象
player - 玩家对象
ctx - Canvas 2D 上下文
W H - 画布宽高

游戏控制

game.addCoin(x, y) - 在指定位置生成金币
game.addLine(type, speed) - 生成线条
game.spawnParticles(x, y, color, count) - 生成粒子
game.setTime(seconds) - 设置游戏时间
game.setLives(n) - 设置生命值
game.addLives(n) - 增加生命值
game.setCoinScore(n) - 设置金币数

玩家控制

player.x player.y - 位置
player.r - 半径
player.setPos(x, y) - 设置位置
player.setColor(r, g, b) - 设置颜色
player.setRadius(r) - 设置半径

绘制

game.drawText(text, x, y, options) - 绘制文字
game.drawRect(x, y, w, h, color) - 绘制矩形
game.drawCircle(x, y, r, color) - 绘制圆形

事件钩子

在代码中定义以下函数即可自动触发：

function onUpdate(dt) {
  // 每帧调用，dt为时间差
}
function onDraw() {
  // 每帧绘制后调用
}
function onCoinCollect() {
  // 收集金币时调用
}
function onDamage() {
  // 受伤时调用
}

示例

// 彩虹玩家
let hue = 0;
function onUpdate(dt) {
  hue = (hue + dt * 100) % 360;
  let c = hslToRgb(hue, 100, 50);
  player.setColor(c[0], c[1], c[2]);
}
function onDraw() {
  game.drawText("Rainbow Mode", W/2, 80, {
    color: "#fff", size: 20, align: "center"
  });
}
